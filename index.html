// --- НОВЫЙ МЕХАНИЗМ ОБЛАКА (JSONBlob) --- async function uploadToCloud() { let id = document.getElementById('syncIdInput').value.trim(); let payload = [...places]; if (userLocation) { payload.push({ id: 'user_live_location', isUserLocation: true, lat: userLocation.lat, lng: userLocation.lng, timestamp: new Date().toLocaleString() }); } try { if (!id) { const res = await fetch('https://jsonblob.com/api/jsonBlob', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' }, body: JSON.stringify(payload) }); if (!res.ok) throw new Error("Сервер отклонил создание"); // Получаем ID из заголовка ответа Location const location = res.headers.get('Location'); id = location.split('/').pop(); localStorage.setItem('ekb_sync_id', id); document.getElementById('syncIdInput').value = id; alert('Создан ваш Sync ID: ' + id + '\n\nСкопируйте его и отправьте друзьям.'); } else { const res = await fetch('https://jsonblob.com/api/jsonBlob/' + id, { method: 'PUT', headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' }, body: JSON.stringify(payload) }); if (!res.ok) throw new Error("Код не найден"); localStorage.setItem('ekb_sync_id', id); alert('Данные и ваша позиция успешно выгружены!'); } } catch (e) { alert('Ошибка синхронизации: ' + e.message); } } async function downloadFromCloud() { let id = document.getElementById('syncIdInput').value.trim(); if (!id) return; try { const res = await fetch('https://jsonblob.com/api/jsonBlob/' + id); if (!res.ok) throw new Error("Неверный код"); markersMap = {}; places = filteredData; localStorage.setItem('ekb_places', JSON.stringify(places)); places.forEach(renderMarker); alert('Загружено!'); closeModal('syncModal'); } else { alert('Файл пуст'); if(friendLoc) { found++; if(friendMarkersMap[f.id]) map.removeLayer(friendMarkersMap[f.id]); const icon = L.divIcon({ html: `
